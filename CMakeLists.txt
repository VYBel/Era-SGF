cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" ERA_SGF_VERSION LIMIT_COUNT 1)
project(EraSGF VERSION ${ERA_SGF_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ERASGF_POCKETBOOK "Build the native InkView .app" OFF)
add_library(erasgf STATIC src/sgf.cpp src/viewer.cpp src/encoding.cpp src/storage.cpp src/render.cpp)
target_include_directories(erasgf PUBLIC include)
target_compile_definitions(erasgf PUBLIC ERA_SGF_VERSION="${PROJECT_VERSION}")
target_compile_options(erasgf PRIVATE -Wall -Wextra -Wpedantic)

if(ERASGF_POCKETBOOK)
    find_path(INKVIEW_INCLUDE_DIR inkview.h)
    find_library(INKVIEW_LIBRARY inkview)
    if(NOT INKVIEW_INCLUDE_DIR OR NOT INKVIEW_LIBRARY)
        message(FATAL_ERROR "InkView headers/library not found; select the PocketBook SDK toolchain")
    endif()
    add_executable(era-sgf src/pocketbook.cpp)
    target_include_directories(era-sgf PRIVATE "${INKVIEW_INCLUDE_DIR}")
    target_link_libraries(era-sgf PRIVATE erasgf "${INKVIEW_LIBRARY}" m pthread dl)
    target_link_options(era-sgf PRIVATE -static-libstdc++ -static-libgcc)
    set_target_properties(era-sgf PROPERTIES SUFFIX ".app")
else()
    add_executable(era-sgf-inspect src/inspect.cpp)
    target_link_libraries(era-sgf-inspect PRIVATE erasgf)
    add_executable(era-sgf-preview src/preview.cpp)
    target_link_libraries(era-sgf-preview PRIVATE erasgf)
    include(CTest)
    if(BUILD_TESTING)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_executable(core-tests tests/core_tests.cpp)
        target_link_libraries(core-tests PRIVATE erasgf)
        add_executable(pocketbook-host src/pocketbook.cpp tests/inkview_stub/inkview.cpp)
        target_include_directories(pocketbook-host BEFORE PRIVATE tests/inkview_stub)
        target_link_libraries(pocketbook-host PRIVATE erasgf)
        add_test(NAME core COMMAND core-tests)
        add_test(NAME files COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration_test.py" "$<TARGET_FILE:era-sgf-inspect>")
        add_test(NAME device-adapter COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/ui_test.py" "$<TARGET_FILE:pocketbook-host>" "$<TARGET_FILE:era-sgf-inspect>")
        add_test(NAME packaging COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/package_test.py")
        set_tests_properties(core files device-adapter packaging PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    endif()
endif()
